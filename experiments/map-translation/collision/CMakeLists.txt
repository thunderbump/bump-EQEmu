cmake_minimum_required(VERSION 3.20)
project(collision_evaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(COLLISION_SANITIZERS "Instrument standalone C++ evaluation" OFF)
get_filename_component(SERVER "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(collision-eval driver.cpp ${SERVER}/zone/raycast_mesh.cpp ${SERVER}/common/memory/ksm.cpp)
target_include_directories(collision-eval PRIVATE shim ${SERVER} ${SERVER}/zone)
if(COLLISION_SANITIZERS)
 target_compile_options(collision-eval PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -O1 -g -fno-omit-frame-pointer -fno-pie)
 target_link_options(collision-eval PRIVATE -fsanitize=address,undefined -no-pie)
endif()
