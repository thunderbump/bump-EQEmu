cmake_minimum_required(VERSION 3.20)
project(map_capture LANGUAGES CXX)
find_package(ZLIB REQUIRED)
find_path(GLM_INCLUDE_DIR glm/vec3.hpp REQUIRED)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(map-capture capture.cpp ${ROOT}/zone/map_geometry.cpp ${ROOT}/common/compression.cpp)
target_include_directories(map-capture PRIVATE ${ROOT} ${GLM_INCLUDE_DIR})
target_link_libraries(map-capture PRIVATE ZLIB::ZLIB)
target_compile_features(map-capture PRIVATE cxx_std_17)
target_compile_options(map-capture PRIVATE -fno-fast-math -ffp-contract=off)
option(MAP_CAPTURE_SANITIZERS "Opt-in ASan/UBSan diagnostics" OFF)
if(MAP_CAPTURE_SANITIZERS)
 target_compile_options(map-capture PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -fno-pie)
 target_link_options(map-capture PRIVATE -fsanitize=address,undefined -no-pie)
 add_executable(map-alignment-control alignment_control.cpp)
 target_compile_options(map-alignment-control PRIVATE -fsanitize=undefined -fno-sanitize-recover=all -fno-pie)
 target_link_options(map-alignment-control PRIVATE -fsanitize=undefined -no-pie)
endif()
