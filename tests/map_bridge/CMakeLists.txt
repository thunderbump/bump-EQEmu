cmake_minimum_required(VERSION 3.20)
project(map_bridge_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# The helper intentionally resolves the maintained crate relative to project root.
set(CMAKE_SOURCE_DIR "${REPO}")
include("${REPO}/cmake/MapRaycaster.cmake")
add_executable(bridge_test bridge_test.cpp "${REPO}/zone/rust_raycast_mesh.cpp")
target_include_directories(bridge_test PRIVATE "${REPO}/zone")
eqemu_link_map_raycaster(bridge_test)

add_executable(bridge_replay replay.cpp "${REPO}/zone/rust_raycast_mesh.cpp" "${REPO}/zone/raycast_mesh.cpp" "${REPO}/common/memory/ksm.cpp")
target_include_directories(bridge_replay PRIVATE "${REPO}/experiments/map-translation/collision/shim" "${REPO}" "${REPO}/zone")
eqemu_link_map_raycaster(bridge_replay)
# Assertions must remain active in Release tests.
target_compile_options(bridge_test PRIVATE -UNDEBUG)
find_path(GLM_INCLUDE_DIR glm/vec3.hpp HINTS /deps ENV GLM_INCLUDE REQUIRED)

find_package(ZLIB REQUIRED)
add_executable(wrapper_test wrapper_test.cpp "${REPO}/zone/map.cpp" "${REPO}/zone/map_geometry.cpp" "${REPO}/zone/rust_raycast_mesh.cpp" "${REPO}/common/compression.cpp")
target_include_directories(wrapper_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/shim" "${REPO}" "${GLM_INCLUDE_DIR}")
target_link_libraries(wrapper_test PRIVATE ZLIB::ZLIB)
target_compile_options(wrapper_test PRIVATE -UNDEBUG)
target_compile_definitions(wrapper_test PRIVATE FMT_HEADER_ONLY)
eqemu_link_map_raycaster(wrapper_test)

# Compile the same production Map logic with its default native engine selection.
add_library(default_map_compile OBJECT "${REPO}/zone/map.cpp")
target_include_directories(default_map_compile PRIVATE "${CMAKE_CURRENT_LIST_DIR}/shim" "${REPO}" "${GLM_INCLUDE_DIR}")
target_compile_definitions(default_map_compile PRIVATE FMT_HEADER_ONLY)
if(EQEMU_SANITIZERS)
  add_executable(sanitizer_controls "${REPO}/tests/sanitizer_controls.cpp")
  foreach(t bridge_test bridge_replay wrapper_test sanitizer_controls default_map_compile)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -g -O1 -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
